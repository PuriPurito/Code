## 2.2.2.8 Массив competences

Для формы оценки компетенций.

Содержит массив объектов, описывающих настройки отображения каждой компетенции из набора в форме, подлежащих оценке. Структура объекта, описывающего компетенцию, следующая:

{

id: “”, // Идентификатор компетенции

name: “”, // Название компетенции

_title: "", // HTML-название компетенци для показа в заголовках_

weight: {}, // Вес компетенции

competence\_block\_id: “”, // Идентификатор блока компетенций, если используется разбивка по блокам, должен присутствовать в массиве competence\_blocks

description: “”, // Описание компетенции

editable: true, // Признак возможности изменить значение

mark: {}, // Оценка по компетенции, текст (только оценка) или объект (оценки и параметры отображения/редактирования)

value: ””, // Числовое значение оценки

comment: “”, // Комментарий к оценке компетенции

has\_comment: false, // Признак возможности ввода/показа комментария

comment\_after\_control: false, // Показывать поле комментария после элемента ввода оценок, по умолчанию (false) поле ввода комментария выводится ниже названия индикатора

control\_below\_title: false, // Показывать элемент ввода оценки ниже названия компетенции/индикатора, по умолчанию элемент ввода расположен справа от названия

mark\_label: ””, // Заголовок поля оценки компетенции при отображении в общей форме (если ввод оценок в отдельном диалоговом окне)

mark\_self: ””, // Значение самооценки (если предусмотрено настройками процедуры)

mark\_self\_label: ””, // Заголовок поля самооценки при отображении в общей форме

mark\_self\_position: 0, // Порядок показа поля самооценки в ряду (руководитель/эксперт...)

mark\_manager: ””, // Значение оценки руководителя (если предусмотрено настройками процедуры)

mark\_manager\_label: ”” // Заголовок поля оценки руководителя при отображении в общей форме

mark\_manager\_position: 1, // Порядок показа поля самооценки в ряду (руководитель/эксперт...)

mark\_expert: ””, // Значение оценки эксперта (если предусмотрено настройками процедуры)

mark\_expert\_label: ”” // Заголовок поля оценки эксперта при отображении в общей форме

mark\_expert\_position: 2, // Порядок показа поля оценки эксперта в ряду (руководитель/эксперт...)

mark\_interview: ””, // Значение оценки собеседования (если предусмотрено настройками процедуры)

mark\_interview\_label: ”” // Заголовок поля оценки собеседования при отображении в общей форме

mark\_interview\_position: 3, // Порядок показа собеседования в ряду (руководитель/эксперт...)

icon\_url: ””, // Пиктограмма для компетенции

class: ””, // Класс, будет добавлен в таблицу стилей контейнера отображающего компетенцию

weight\_view: “”, // Параметр показа веса компетенции (none – не показывать вес)

mark\_view: “”, // Параметр показа оценки по компетенции (none – не показывать оценку) – для способа оценки компетенции по индикаторам

scale: \[\], // Шкала оценки компетенции

indicators: \[\] // Массив объектов, описывающих параметры отображения индикаторов, входящих в состав компетенции

check\_require\_on\_select: false // - предупреждать об обязательности комментария сразу после выбора значения оценки, имеет приоритет над одноимеенным параметр в конфиге анкеты

}
