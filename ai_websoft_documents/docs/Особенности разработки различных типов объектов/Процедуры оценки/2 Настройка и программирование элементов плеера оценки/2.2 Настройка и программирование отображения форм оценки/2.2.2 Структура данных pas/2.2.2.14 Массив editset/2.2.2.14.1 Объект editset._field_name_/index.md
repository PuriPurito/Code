## 2.2.2.14.1 Объект editset.<field_name>

Для каждого поля, входящего в состав блока, создается отдельный объект с именем поля, который описывает значение поля и настройки его показа/редактирования.

Есть 4 зарезервированных названия объектов для показа элементов формы, не связанных с полями формы:

\- header – текст, размещаемый в верхней части формы, сразу под заголовком окна

\- divider – разделитель (сплошная линия)

\- instruction – текст, размещаемый в качестве инструкции по заполнению полей формы

\- richtext – текст, размещаемый в произвольном месте формы

Элементы header|instruction|richtext – по сути, один и тот же элемент (произвольный текстовый блок, с поддержкой HTML\-развертки) и отличаются друг от друга только стандартным стилем, применяемым к блоку по умолчанию, условно говоря: header – полужирный, instruction – красный, richtext – без эффектов.

Если на форме необходимо использовать несколько стандартных элемекнтов оформления, то можно использовать суффиксные названия - к зарезервированному слову добавляется суффикс с номером по порядку (divider\_0, divider\_1, header\_1, richtext\_1 и т.п.)

Других стандартных или зарезервированных полей в форме общей оценки нет.

Структура описания поля следующая:

{

control\_type: ””, // Тип поля, возможные значения:

//            group – специальный тип – группа полей, объединенная в контейнере с возможностью свернуть (оставив только название) или развернуть содержимое контейнера

//            text – поле для ввода и показа текста

//            number – числовой ввод, показ – как текст

//            date – ввод даты, показ – как текст

//            textarea – ввод нескольких строк текста, показ – как текст

//            select – выбор значения из выпадающего списка, показ – как текст

//            list – выбор нескольких значений в окне выбора слева, с переносом выбранного в окно справа, показ – как список в несколько строк текста

//            checkbox – установка «галки» в квадрате, показ - аналогично

//            slider – выбор значения путем перемещения указателя по линейке, показ – выбранное значение

//            spinner – поле числового ввода с возможностью изменения значения мышью в полосе прокрутки

//            file – выбор файла, при показе – возможность скачать/просмотреть содержимое

//            url – ввод текста ссылки, при показе – возможность перейти по клику на ссылке

//           foreign\_elem - выбор объекта из каталога

label: “”, // заголовок (текстовое название) поля

class: “”, // класс, который будет добавлен в список стилей контейнера поля

_desc\_form: "", // Текст всплывающего сообщения (tooltip/title), будет показан при наведении мыши на заголовок поля в окне редактирования_

_desc\_from\_class: "", // дополнительный класс для сообщения_

_desc\_view: "",_ _// Текст всплывающего сообщения (tooltip/title), будет показан при наведении мыши на заголовок поля в списке/таблице_

_desc\_view\_class: "",_ _// дополнительный класс для сообщения_

_desc\_value: "",_ _// Текст всплывающего сообщения (tooltip/title), будет показан при наведении мыши на элемент ввода в окне редактирования_

_desc\_value\_class: "",_ _// дополнительный класс для сообщения_

editable: true, // признак возможности изменить значение поля

required: false, // признак обязательности - необходимости заполнить/выбрать значение

visible: true, // признак видимости – возможность создания скрытых полей

visible\_in: “”, // область видимости, строка значений, разделенных точкой с запятой. Значения:

//            title – значение поля показывается в заголовке окна редактирования/просмотра цели

//            view – значение поля показывается в основной форме (плашка или ячейка таблицы)

//            form – поле показывается для просмотра/редактирования в диалоговом окне показа/редактирования цели

group: “”, // Группа, в составе которой будет показано поле (должно совпадать с именем одного из объектов с типом group, причем этот объект должен быть описан до появления поля, ссылающегося на него)

expand\_in: “”, // для полей типа group – развернутое состояние группы в указанных областях (анлогично visible\_in – одно из двух или оба значения view;form). По умолчанию группа свернута во всех областях. Например expand\_in: “form” – группа будет свернута в основной форме (на плашке цели), но развернута при открытии окна диалога редактирования

layout: “”, // для полей типа checkbox или radio – ориентация элементов контрола (радиокнопок или чекбоксов) – col (размещение в колонках по горизонтали – одна строка, несколько элементов) | row (размещение в строках по вертикали – несколько строк, по одному элементу в каждой)

spot\_position: “”, // Положение радио/чекбокса относительно текста выбираемого значения:

// left – элемент слева, текст справа

// right – элемент справа, текст слева

// top – элемент выше, текст под элементом

// bottom – элемент ниже, текст над элементом

spot\_type: “”, // тип элемента выбора – plain (круглые элементы) | square (квадратные элементы)

value\_position: “”, // для поля типа slider – положение области, где показывается текущее установленное значение left|right|top|bottom

data\_src: “”, // для элементов со списками (list/select/checkbox/slider/spinner) – если используется общий массив данных (из объекта dataset) - название справочника, хранящего данные для выбора

scale: { // задание массива данных для выбора (если не задан data\_src)

                value: \[ // массив значений для выбора

                                {

                                id: “”, // сохраняемое значение

                                name: “”, // показываемый текст для значения

                                desc: “”, // описание, показывается при наведении мыши на значение

                                class: “”, // класс, который будет добавлен в таблицу стилей контейнера для данного значения (возможность, например, раскрашивать значения в списке выбора)

                                disabled: false // признак неактивности, если true – значение будет показано в списка, но его нельзя будет выбрать

                                }

                                \]

                },

placeholder: “”, // «заглушка» - текст, который будет показан в поле, если не выбрано значение

default: “”, // значение по умолчанию, устанавливается при создании цели

_default\_name: "" //_  _для поля типа foregn\_elem - название объектов по умолчанию, через точку с запятой_

step: “”, // для поля типа spinner – шаг смены значения при клике на полосе прокрутки

min_: 0, // для полей типа_ number _и_ spinner _– минимально возможное значение ввода, для полей типа date - минимально возможная дата выбора из календаря (в формате ISO 8601 или dd.mm.yy)_

max_: 100 // для полей типа_ number _и_ spinner _\- максимально возможное значение ввода, для полей типа date - максимально возможная дата выбора из календаря_ _(в формате ISO 8601 или dd.mm.yy)_

value: “” // Значение поля

 file\_name: “”, // для полей типа file – название файла

 file\_type: “”, // для полей типа file – тип файла (doc, txt, pdf и т.п.) – для пиктограммы

  file\_url: “”, // для полей типа file – url на скачивание файла

  file\_size: “”, // для полей типа file – размер файла в байтах

name: "" // _для поля типа foregn\_elem - название выбранных объектов, через точку с запятой_

_catalog\_name: "" // для поля типа foregn\_elem - название катлога_

_xquery\_qual: "" // для поля типа foregn\_elem - дополнительное условие XQuery, относительно $elem_

_multi\_select: true|false // для поля типа foregn\_elem - признак множественного выбора_

_show\_all: true|false // для поля типа foregn\_elem - учитывать условия огранияения (по текущему пользователю) на объекты типа collaborator, subdivision. org_

_page\_url: "" // для поля типа foregn\_elem - адрес кастомной страницы диалога, в которой будет осуществлятся выбор объекта_

}
