## 2.2.2 Структура данных pas

Представляет из себя массив объектов “pa”, каждый из которых описывает отдельную оценочную форму. Структура объекта “pa” следующая:

{

id: “6363083350092180177”, // Идентификатор документа оценочной формы

type: “competence\_appraisal”, // тип оценочной формы – (competence\_appraisal, activity\_appraisal, activity\_plan, staffrating, result)

index: 2, // индекс (поле в форме, переносится из настроек процедуры оценки для данного типа и статуса)

is\_done: false, // признак завершенности формы

is\_processed: true, // признак начала работы с формой

status: “self”, // статус формы – самооценка (self), оценка руководителем (manager), оценка подчиненными (staff), оценка коллегами (coll), оценка экспертом (expert), оценочное собеседование (interview)

editable: true, // признак возможности редактирования формы

type\_title: “Самооценка компетенций”, // заголовок формы (ярлык закладки)

workflow\_state: “3”, // этап документооборота

overall: 100, // Итоговая оценка по форме

sum\_weight: 100, // Сумма весов целей или компетенций (в зависимости от формы)

print\_url: “/view\_print\_form.html…”, // Адрес для открытия печатной формы – ссылка, открывающая при нажатии кнопку «Печать» в панели ДО форме оценки

person: {}, // Объект типа person, содержащий информацию об оцениваемом сотруднике (поле person\_id)

expert: {}, // Объект типа person, содержащий информацию об оценивающем (поле expert\_id)

boss: {}, // Объект типа person, содержащий информацию о руководителе (поле boss\_id плана оценки)

custom\_experts: \[\], // Массив объектов типа person, содержащих информацию об экспертах

comment: “”, // Строка, содержащая текущий комментарий пользователя на данном этапе ДО

custom\_comments: \[\], // Массив объектов – все комментарии к форме

wfcomment: “”, // Текст подсказки, показываемой в верхней части формы

wfcomment\_title: “”, // Заголовок блока, в котором размещается текст подсказки

wfcomment\_state: “”, // Состояние блока подсказки: collapsed - блок свернут, другое – развернут

wfcomment\_class: “”, // Название класса, будет добавлено в таблицу стилей блока подсказки

class: “”, // Название класса, будет добавлено к верхнему элементу контейнера для данной формы

view\_type: “”, // Режим отображения: для формы компетенций – simple | list | table | tiles | unified\_table | accordion | random\_table, для формы целей – tree | table

view\_type\_level: “”, // только для формы целей в режиме table – уровень цели (начиная с 0), с которого включается режим таблицы - т.е. все дочерние цели ниже этого уровня будут отображаться в виде таблицы. если указан уровень "-1" - в форме возможен только один уровень целей и он отобразится в виде таблицы

overall\_view: “”, // Режим отображения виджета итоговой оценки – top | bottom | none

overall\_view\_class: “”, // Класс виджета итоговой оценки

overall\_view\_title: “”, // Заголовок виджета итоговой оценки

_overall\_view\_old\_style: true. // Режим совместимости с предыдущим вариантом виджета итоговой оценки - когда блоки итогов и весов разделены_

weight\_view: “”, // Режим отображения виджета суммы весов (компетенций или целей) формы - top | bottom | none

tiles\_in\_row: 4, // Количество плашек в строке формы оценки компетенций (режим Плашки)

reload: false, // Признак обновления формы при переходе (клике) по закладкам, по умолчанию – false (все формы-закладки загружаются один раз, при клике из дерева оценки) , если true – при каждом переходе на форму она обновляется с сервера

msg: {}, // Объект, содержащий данные для показа окна сообщения после отрисовки формы, в упрощенном варианте – текст сообщения (строка)

msg\_class: “”, // Класс, применяемый для окна сообщения в упрощенном варианте (только текст)

config: {}, // Объект конфигурации формы

wfparameters: {}, // Объект параметров отображения элементов формы

wfbuttons: \[\], // Массив объектов, содержащих описания доступных кнопок документооборота <div class="wt-btn-icon-container">

dataset: {}, // Набор объектов, содержащий массивы справочников общего пользования

competences: \[\], // Массив объектов, содержащих информацию об оцениваемых компетенциях

competence\_blocks: \[\], // Массив объектов, содержащих информацию о блоках компетенций

n\_allowed: false, // Признак наличия варианта отказа от оценки компетенции или индикатора

n_\__title_:_ false_, // Описание значения для варианта отказа от оценки - по умолчанию "Не могу оценить"_

n_\__name_:_ false_, // Текст значения для варианта отказа от оценки - по умолчанию - "N"_

objectives: \[\], // Массив объектов, содержащих данные о целях/задачах/КПЭ и т.п.

objective\_blocks: \[\], // Массив объектов, содержащих информацию о блоках задач/целей

obj\_weight: {}, // Массив данных для интерфейса редактирования весов задач в отдельном окне

can\_delete: false, // Признак возможности удалять задачи (в целом по форме)

can\_add\_child: true, // Признак возможности добавлять задачи (в целом по форме)

can\_translate: false, // Признак возможности транслирования задач (будет подгружен список подчиненных)

viewset: {}, // Объект, содержащих данные о показе дополнительных виджетов на форме

editset: \[\], // Массив объектов, описывающих настройки блоков ввода/показа данных для формы общей оценки, дополнительных вопросв для форм целей и компетенций

integral\_mark: 0, // Интегральная оценка (для формы общей оценки)

save\_obj\_id: “” // Атрибут для передачи идентификатора цели, которая была изменена и данные по которой сохраняются

}

Обработчик формирует объект “pa” для каждой оценочной формы и передает его в код дополнительного шаблона, прописанного в процедуре оценки для данного типа оценки и статуса формы.

Объект доступен в коде шаблона, как глобальная переменная oData.data.

К изменению доступны любые свойства объекта.

Все атрибуты и подчиненные объекты не являются обязательными, т.е. в не обязательно присутствуют в глобальном объекте, формируемым обработчиком. Программисту необходимо проверять наличие любого объекта перед его модификацией - oData.data.HasProperty(“dataset”).
