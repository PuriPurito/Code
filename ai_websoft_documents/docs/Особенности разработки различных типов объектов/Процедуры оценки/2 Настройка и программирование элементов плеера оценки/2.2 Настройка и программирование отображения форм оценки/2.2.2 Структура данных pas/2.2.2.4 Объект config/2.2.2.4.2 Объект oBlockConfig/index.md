## 2.2.2.4.2 Объект oBlockConfig

Объект предназначен для детального описания настроек работы режима блокировки. Если объект не задан - работают параметры по умолчанию.

oData.data.config.oBlockConfig = {  
    bCheckOnFree: true, // по умолчанию true - включить проверку состояния формы в режиме чтения. Если вклбчено, браузер постоянно опрашивает сервер о том, в каком статусе находится форма, даже если пользователь не вклбчает режим редактирования  
    iBlockCheckInterval: 20000, // по умолчанию 60000 - период, в миллисекундах, с которым браузер производит проверку состояния формы  
    iBlockInterval: 5, // по умолчанию 20 - период, в минутах, на который включается режим блокировки формы  
    bShowNoSave: true, // по умоляанию false, флаг - выводить ли дополнительное (помимо сообщения, что время истекло) сообщение, если у пользователя открыт диалог редактирования цели в момент истечения периода редактирования. если false - то пользователь видит инфо, что время истекло, далее при нажатии "Сохранить" - просто ничего не сохраняется, если true - то при каждой попытке сохранить выводится дополнительное окно предупреждения  
    sNoSaveText: "Нельзя сохранить...", // текст в окне дополнительного сообщения, по умолчанию - "Время, отведенное на редактирование, истекло. Если Вы не успели сохранить данные, скопируйте изменения, закройте диалог без сохранения и заново активируйте режим редактирования"  
    bOnlyOnePa: false, // флаг, устанавливающий возможность блокировать на редактирование только одну анкету из процедуры в один момент - т.е., если у пользователя есть несколько закладок с доступными анкетами и в них возможен запрос на блокировку для редактирование, то установка данного флага приведет, при попытке запросить редактирование второй анкеты, к отказу и появлению сообщения.  
    sOnlyOnePaText: "Разрешена блокировка только одной анкеты", // текст информационного сообщения при попытке заблокировать вторую анкету, при запрете на это, по умолчанию - "Разрешена блокировка только одной анкеты в один момент. Для редактирования текущей анкеты необходимо снять блокировку с &quot;<NAME>&quot;" - может содержать подстановку <NAME>, вместо которой в текст сообщения будет подставлен заголовок ярлыка той уже заблокированной анкеты  
    bShowAlertsAccrossCurrentPa: false, // показывать сообщения только по блокировкам всех анкет - используется, если есть несколько закладок и пользователь, заблокировав одну анкету на редактирование, переходит на другую закладку. при этом на начальной, заблокированной анкете, происходит какое-то событие (заканчивается период редактирования и т.п.). если параметр не задан или false - сообщение при нахождении на другой закладке, не появится, если true - то сообщение о событии на другой закладке покажется на текущей.  
    bShowGoBtnOnAlert: false, // показывать кнопку перехода в целевую анкету, если предыдущий парамет true и сообщение показывается на другой закладке  
    sShowGoBtnText: "Перейти в анкету", // текст кнопки перехода в целевую анкету  
    oNotifyBtn: { // настройки уведомлений в режиме блокировки  
        bShowOnFree: false, // по умолчанию false - показывать или нет кнопку "Уведомлять о состояниянии формы" в режиме "свободной" формы - если true, то пользователь даже не входя в режим редактирования, может подписаться на уведомления о смене состяния формы (блокировка/разблокировка)  
        bShowOnBlock: true, // по умолчанию true - показывать или нет кнопку "Уведомлять о состояниянии формы" в режиме, когда форма заблокирована другим пользователем  
        bShowCancel: true, // по умолчанию true - показывать или нет кнопку "Отменить уведомления" - т.е. иожет ли пользователь, подписавшись на уведомления о смене статуса, потом отказаться от них  
        sBtnText: "Уведомлять об изменении состояния", //текст кнопки "Уведомить..."  
        sCancelText: "Отменить уведомления" // текст кнокпи "Отменить уведомления"  
    },  
    oRequestBtn: {  
         bShowSeparately: true, // показывать отдельную кнопку "Запросить редактирование", по умолчанию действие предлагается при нажатии на плашку с оставшимся временем  
         bConfirm: true, // для отдельной кнопки - показывать диалог подтверждения, или сразу отправлять запрос на сервер  
         sConfirmText: "Текущее состояние: <msg-title><br/>Запросить снятие блокировки?", // текст диалога запроса, <msg-title> - текущая информация, кто и насколько блокировал  
         sBtnText: "Запросить блокировку (освободить форму)", // текст отдельной кнопки  
         sConfirmOKText: "ОК, запросить", // текст кнопки ОК в диалоге  
         sConfirmCancelText: "Нет, отменить" // текст кнопки Отмена в диалоге  
     },  
    oMyBtn: {  
          sConfirmText: "Форма в настоящий момент открыта Вам в режиме редактирования. Вы можете продлить период редактирования или освободить форму для других пользователей", // текст сообщения при клике на плашку с информацией о блокировке  
          sBtnTextOk: "Продлить период", //текст для кнопки действия продления периода редактирования  
          sBtnTextCancel: "Завершить редактирование", // текст кнопки действия завершения редактирования  
          bHideOk: false, // скрывать кнопку продления периода  
          bHideCancel: false, // скрывать кнопку завершения редактирования  
          bNoClick: false // не отрабатывать нажатие на плашку с информацией (не показывать окно с информацией и кнопками)  
   },  
   oBlockBtn: {  
           sText: "Осталось <TTL> мин.", // текст плашки (красного цвета) с информацией, когда форма открыта на редактирование другим сотрудником,   
           sTip: "Форма в настоящий момент открыта в режиме редактирования сотрудником: <FIO>", // текст информации (всплывыающий текст) о том, что форма открыта на редактирование другим сотрудником,   
    },  
   oFreeBtn: {  
           sTextTitle: "Внимание", //заголовок окна сообщения об истечении срока редактирования  
           sTextEnd: "Период редактирования закончился." // сообщение об истечении срока  
   },  
    events :// массив детальрных настроек работы уведомления для различных событий формы  
    {  
        clear: { // событие прекращение режима блокировки (пользователь прекратил редактирование) - уведомления, отправляются ожидающим пользователям  
            notification: true, // флаг - отправлять ли почтовое уведомление  
            notification\_id: "clear\_ipr\_notif", // код типа уведомления  
            push: true, // открывать ли окно сообщения в форме пользоватпеля (работат, если плеер загружен и форма открыта)  
            push\_text: "Форма доступна для редактирования" // текст уведомления  
        },  
        warning: { // до окончания периода редактирования формы осталось мало времени - уведомление тому, кто редактирует  
            notification: true,  
            notification\_id: "warning\_ipr\_notif",  
            push: true,  
            push\_text: "До окончания периода редактирования осталось <TTL> минут",  
            nminutes: 15 // время, в минутах, остащееся до остановки режима редактирования,  
            "update": true, // предлагать продлить блокировку в диалоге окна предупреждения  
            "ok\_text": "Спасибо, понял", // текст кнопки ОК в диалоге, по умолчанию "Больше не предупреждать"  
            "update\_text": "Еще времени, не успеваю" // текст кнопки продления периода в диалоге, если update = true,  
            title\_text: "Внимание", //  текст заголовка окна с предупреждением об истечении времени  
            cancel\_text: "Осталось <TTL> мин." // текст информации о скором завершении периода редактирования  
        },  
        stop: { // период редактирования закончен для текущего пользователя - уведомление тому, кто редактирует  
            notification: true,  
            notification\_id: "stop\_ipr\_notif",  
            push: true,  
            push\_text: "Форма более недоступна для редактирования"  
        },  
        update: { // продление периода редактирования - уведомление ожидающим  
            notification: true,  
            notification\_id: "update\_ipr\_notif",  
            push: true,  
            push\_text: "Форма более недоступна для редактирования"  
        },  
        set: { // включение режима редактирования по команде пользователя - уведомления ожидающим  
            notification: true,  
            notification\_id: "set\_ipr\_notif",  
            push: true,  
            push\_text: "Форма заблокирована сотрудником <fio> до <period>"  
        },  
        request: { // запрос на редактирование - уведомление тому, кто редактирует. если элемент есть, то можно запросить редактирование (т.е. тому, кто блокировал, приходить письмо или высплывает окно сообщения, что другой пользователь тоже хочет внести изменения)  
            notification: true,  
            notification\_id: "request\_ipr\_notif",  
            push: true,  
            push\_text: "<fio> запросил редактирование формы"  
        }  
    }  
}

При отправке уведомления вызывается функций tools.create\_notification( template, user, text )

template - код типа уведомления, заданный для события

user - ID пользователя, подписавшегося на событие

text - JSON-строка, содержит:

      pa\_id - ID анкеты

     state - статус анкеты ( free - анкета свободна, иное - занята)

     ttl - время, в минутах, оставшееся до окончания блокировки

     user\_id - ID пользователя, который в настоящий момент блокирует анкету

     fio - ФИО пользователя, который блокирует
