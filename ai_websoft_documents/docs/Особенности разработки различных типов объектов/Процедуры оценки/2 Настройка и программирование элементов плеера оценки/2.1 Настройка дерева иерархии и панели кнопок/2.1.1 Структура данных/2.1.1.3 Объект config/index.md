## 2.1.1.3 Объект config

Содержит набор атрибутов и объектов, описывающих настройки интерфейса показа дерева оценки.

По умолчанию обработчик создает пустой объект. Все атрибуты создаются только программным кодом дополнительного шаблона дерева оценки.

Доступен в коде шаблона в переменной oData.config.

Структура объекта:

// параметры конфигурации интерфейса дерева оценки

left\_panel\_state: expand|collapse - состояние левой панели (с пунктами меню) по умолчанию - развернута или свернута

"buttons": \[ \], //массив описаний дополнительных кнопок левой панели плеера

"btn\_users": true, // флаг показа/скрытия кнопки «Иерархия подчиненности», по умолчанию - да

"btn_\__users\_title": "", // текст под кнопкой, по умолчанию _«Иерархия подчиненности»_

_"btn\_sheet\_title": "", // текст позиции меню "Оценочные формы"_

"btn\_objectives": true, //флаг показа кнопки «Дерево целей», по умолчанию - да

_"_btn_\__objectives\_title_": "", // текст под кнопкой, по умолчанию_ _«__Дерево целей__»_

"btn\_logs": true, //флаг показа кнопки «Журнал действий», по умолчанию - да

_"_btn_\__logs\_title_": "", // текст под кнопкой, по умолчанию_ _«__Журнал действий__»_

"btn\_iframe": true, //флаг показа дополнительных кнопок, по умолчанию зависит от наличия непустого массива buttons

"tree\_person\_filter": \[\], //объект описания фильтра (набора селекторов, расположенного выше дерева оценки)

“expand\_all\_tree”: true, // флаг, показывающий, разворачивать ли все дерево оценки при первом открытии плеера, по умолчанию развернут только первый и второй уровни (формы текущего пользователя и формы подчиненных), остальные уровни свернуты (но могут быть развернуты нажатием на стрелки в интерфейсе). Параметр не работает, если в фильтрах (см описание ниже) выбраны значения, отличные от all (Все формы)

“pa\_id”: “6466355517103872284”, // Идентификатор анкеты, открываемой по умолчанию, если непустой и анкета доступна пользователю, то в после загрузки плеера дерево скрывается, а открывается оценочная форма с требуемым идентификатором

“wta\_class”: “competence\-360”, //класс, общий для интерфейса плеера, если не пусто – данный класс добавляется к верхнему элементу HTML\-объектов плеера (BODY)

“wfcomment\_users”, // Текст комментариев/описания, которое показывается в раскрывающейся секции над деревом оценки (возможность дать быструю подсказку по работе с интерфейсом). Если атрибут отсутствует или пуст, блок подсказки не показывается.

“wfcomment\_title\_users”, // заголовок раскрывающегося блока с подсказкой. Если пусто – выводится дефолтное название «Полезная информация».

“wfcomment\_state\_users”, // состояние, по умолчанию, блока подсказки. Если выставлено значение “collapsed” – блок свернут, любое другое или атрибут отсутствует – блок развернут.

“wfcomment\_class\_users”, // класс из таблицы стилей, который будет применен к данному блоку.

“tree\_header\_label”, // заголовок окна с деревом оценки. Если этот атрибут пуст – выводится заголовок по умолчанию «Иерархия подчиненности».

“wfcomment\_tasks”, // Текст комментариев/описания, которое показывается в раскрывающейся секции над деревом целей (возможность дать быструю подсказку по работе с интерфейсом). Если атрибут отсутствует или пуст, блок подсказки не показывается.

“wfcomment\_title\_tasks”, // заголовок раскрывающегося блока с подсказкой. Если пусто – выводится дефолтное название «Полезная информация».

“wfcomment\_state\_tasks”, // состояние, по умолчанию, блока подсказки. Если выставлено значение “collapsed” – блок свернут, любое другое или атрибут отсутствует – блок развернут.

“wfcomment\_class\_tasks”, // класс из таблицы стилей, который будет применен к данному блоку.

“tasks\_header\_label”, // заголовок окна с деревом целей. Если этот атрибут пуст – выводится заголовок по умолчанию «Каскадирование целей».

// параметры конфигурации заголовка оценочных форм, общие для всей процедуры

“flow\_sides”, // признак показа блоков справа и слева от линейки этапов ДО, может иметь значения = none (скрыты оба блока), left (только левый блок с датой начала), right (только правый блок с датой окончания), both (оба блока), если атрибут отсутствует, значение по умолчанию – both, если установлены дата начала и окончания процедуры или right|left|none, если не установлена одна или обе даты. При скрытии блока ему устанавливается стиль CSS visibility:hidden

“hide\_flow\_left”: false, // флаг выделения/скрытия левого блока (с датой начала) в линейке этапов ДО формы. При наличии значения true в атрибуте блоку устанавливается атрибут wt-opacity=0.

“hide\_flow\_right”: false, // флаг выделения правого блока в линейке этапов. Действие аналогично предыдущему атрибуту, но для правого блока.

// При дефолтной настройке таблицы стилей оба параметра flow\_sides и hide\_flow дают практически одинаковый результат (стандартный стиль \[wt-opacity='0'\] {opacity:0}) – скрытие одного или обоих блоков с датами процедуры

“welcome\_string”: “”, // текст приветствия в верхней части заголовка формы (над названием процедуры)

“assessment\_title”: “”, // название процедуры (в заголовке оценочной формы)

“assessment\_start\_date”: “”, // дата начала процедуры (слева от линейки этапов ДО)

“assessment\_rest\_left”: “”, // текст под датой начала процедуры

“assessment\_end\_date”: “”, // дата окончания процедуры (справа от линейки этапов)

“assessment\_rest\_right”: “” // текст под датой окончания процедуры

sLogoAction: действие при нажатии на логотип в левом верхнем углу

     none  - нет действия

     home - перейти на главную страницу портала

     close - закрыть окно плеера оценки

     back - вернутся назад в истории браузера

    url - перейти по конкретному указанному адресу (по умолчанию)

sLogoConfirm - текст сообщения, если не пусто - выдается запрос Да/Нет с присланным текстом, по умолчанию пусто

sLogoUrl - адрес для перехода, в случае, если тип действия url

sPageTitle - заголовок панели с закладками анкет, по умолчанию "Оценочные формы"
