## 2.1.1.2 Массив pas

Содержит набор объектов, описывающих данные оценочных форм для показа в дереве.

_Массив доступен для изменения разработчику в программном коде кастом-шаблона дерева в виде переменной **oData.****data**_

Структура элемента массива:

{

"id": "6363083350092180178", //идентификатор анкеты

                "person": { // объект данных оцениваемого сотрудника

                                "id": "1105387902724063510", // идентификатор

                                "fullname": "Анисимов Геннадий Федорович", //ФИО

                                "position": "Директор департамента по работе с персоналом", //Должность

                                "icon\_url": "/FACES/M/IS945-082.jpg" //Фото

                },

                "expert": { // данные оценивающего (expert\_person\_id в анкете)

                                "id": "1105387902724063525",

                                "fullname": "Тихомиров Герман Вениаминович ",

                                "position": "_Генеральный директор_",

                                "icon\_url": "/FACES/M/IS945-044.jpg "

                },

                "status": "manager", //_статус анкеты_

                "type": "staffrating", //_тип оценки_

                "is\_main": false, //признак "главной" формы - анкета будет открыта, если несколько форм/закладок, если такого признака нет - откроется первая форма

                "is\_done": true, //признак заполненности формы

                "is\_processed": true, //признак начала работы (устанавливается при первом сохранении данных формы с портала)

                "has\_children": false, //в режиме инкрементальной загрузки - признак наличия форм нижних уровней иерархии (которые нужно загрузить с сервера при разворачивании)

                "workflow\_state": "2", //Код этапа ДО

                "workflow\_set": "", //набор данных (датасет) со списком этап для данной формы

                "index": 1, //Индекс (порядковый номер) анкеты, устанавливается при генерации формы на основе настроек процедуры, в зависимости от типа оценки и статуса анкеты

                “class”: “simple-red-tile”, //класс, присваиваемый плашке (блоку анкет), по умолчанию атрибут отсутствует, выставляется только программным кодом шаблона дерева

                "boss": { //данные руководителя (поле boss\_id в документе плана оценки)

                                "id": "1105387902724063525",

                                "fullname": "Тихомиров Герман Вениаминович",

                                "icon\_url": "/FACES/M/IS945-044.jpg"

                },

                "no\_click": false // признак, что данный узел является виртуальным, т.е. это не анкета, а, например, узлы в цепочке руководителей, которую хочется показать, но при этом при клике на узел не надо открывать анкеты и не надо для таких узлов показывать статус анкеты

                "add\_text": "" // текст, который будет показан в плашке формы, перед линейкой этапов ДО

}
