## Пример программного формирования настраиваемого отчета (выполненные задания)

В данном примере подробно разобраны основные подходы к программированию настраиваемых отчетов.  
В частности, подробно рассмотрены следующие аспекты программирования отчетов:  
_\- формирование и значение первичного ключа;  
\- описание программирования колонок отчета._

Пакет с описываемым отчетом прилагается (см вкладку **Файл**).

// Отбираем все записи из каталога learning\_task\_results (Выполненные задания)  
xarrResp = XQuery("sql: select \* from learning\_task\_results");  
// Запрос типа SQL недоступен при использовании СУБД XML (внутреннего хранилища).   
// В этом случае можно использовать следующий запрос: xarrResp = XQuery("for $elem in learning\_task\_results return $elem");  
  
// Формируется результирующий массив aRes  
aRes = new Array();  
  
// Выполняется обход отобранного массива выполненных заданий  
for( respElem in xarrResp )  
{  
      // Для настраиваемого отчета формируем новый объект (характеризующий выполненное задание)  
      r = new Object();  
      // Задаем первичный ключ (ID сотрудника, выполнившего задание).   
      // При выполнении щелчка мышью по строке настраиваемого отчета будет открываться карточка сотрудника  
      r.PrimaryKey = respElem.person\_id;  
  
      // Если необходимо, чтобы при выполнении щелчка мышью по строке настраиваемого отчета открывалась карточка задания, нужно указать первичный ключ следующим образом:  
      // r.PrimaryKey = respElem.id;  
  
      // Заполняем поля объекта  
      r.id = respElem.id + "";  
      r.person\_id = respElem.person\_id + "";  
      // Добавляем объект в результирующий массив  
      aRes.push(r);  
}  
  
// Формируем столбцы настраиваемого отчета.  
// Предварительно столбцы рекомендуется очистить.  
columns.Clear();  
  
// Создается первый столбец настраиваемого отчета (ID выполненного задания).  
column = columns.AddChild();  
// Первый столбец имеет тип f(x)  
column.flag\_formula = true;  
// Заголовок столбца - id  
column.column\_title = "id";  
// Значения данных - идентификаторы выполненных заданий  
column.column\_value = "ListElem.id";  
// Тип данных первого столбца настраиваемого отчета - строка  
column.datatype = "string";  
  
// Создается второй столбец настраиваемого отчета (ID сотрудника, выполнившего задание).  
column = columns.AddChild();  
// Второй столбец также имеет тип f(x)  
column.flag\_formula = true;  
// Заголовок столбца - person\_id  
column.column\_title = "person\_id";  
// Значения данных - идентификаторы сотрудников  
column.column\_value = "ListElem.person\_id";  
// Тип данных второго столбца настраиваемого отчета - строка  
column.datatype = "string";  
  
// Передача результирующего массива в настраиваемый отчет  
return aRes;
