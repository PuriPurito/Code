## Агент "Изменение продолжительности времени сдачи тестов сотрудником"

Рассмотрим подробнее агент, выполняющий изменение параметра объекта. Данный агент применяется в интерфейсе Администратора. Он находит указанного сотрудника и изменяет параметр **max\_end\_date** (_Дата планируемого завершения_) (устанавливается планируемая продолжительность сдачи всех тестов данного сотрудника, равная 180 дням).

Разберем программный код агента. Комментарии выделены красным шрифтом (текст комментария относится к блоку кода, расположенного выше него), код шаблона - синим.

cFullName = 'Васильева Людмила Петровна'; // ФИО сотрудника  
cUsage\_date = 180; // планируемая продолжительность сдачи всех тестов данного сотрудника  
oCollab1 = tools.get\_doc\_by\_key ( 'collaborator', 'fullname',  cFullName);  
alert ( 'Найден сотрудник ' + oCollab1.TopElem.fullname + ' с идентификационным номером ' + oCollab1.DocID ); // на экран выводится информация об отобранном сотруднике  
iID = oCollab1.DocID;  
  
// найти все незавершенные тесты для отобранного сотрудника  
\_query\_str = "for $elem in active\_test\_learnings where $elem/person\_id=" + XQueryLiteral( iID ) +" return $elem";  
arrActiveTest = XQuery(\_query\_str);  
  
for ( oElem in arrActiveTest) // обработка элементов массива  
{  
     doc = OpenDoc( UrlFromDocID( oElem.id )); // открытие документа по ID  
     doc.TopElem.max\_end\_date = DateOffset (doc.TopElem.start\_usage\_date, cUsage\_date\*24\*60\*60);  // изменение свойства документа max\_end\_date (Дата планируемого завершения)  
     alert( oElem.assessment\_name + ' - окончание обучения:\\n' + doc.TopElem.max\_end\_date );  
     doc.Save(); // сохранение измененного документа в базу  
}
