## ArrayOptFirstElem

      Возвращает первый элемент заданного массива или второй аргумент, если массив пустой.  
      Если массив не содержит ни одного элемента, функция а второй аргумент отсутствует, то функция возвращает _undefined_.  
      Если массив-аргумент отсутствует или не определен (_undefined_), то выдается ошибка с остановкой выполнения кода.

_Синтаксис:_  
      **ArrayOptFirstElem (<array>\[, <value>\])**

_Аргументы:_  
      _<array> (обязательный)_  
      Тип: **Массив**. Исходный массив.  
      _<value> (необязательный)_  
      Тип: **Строка, число, объект или массив**. Значение, которое возвращает функция, если из исходного массива не удалось получить первый элемент (например, если исходный массив пустой). 

_Возвращаемое значение:_  
      Тип: **Элемент массива** или др. Отобранный элемент массива или значение аргумента <value>. Результат действия функции.

_Пример 1:_  
      _try {_  
            _\_query\_str = "for $elem in collaborators where contains($elem/fullname, 'ин') return $elem";  
            personArray = XQuery(\_query\_str);  
            // personArray является массивом объектов Сотрудник, отобранных запросом. Запрос отбирает всех сотрудников, в состав фамилии, имени или отчества которых входит подстрока 'ин'.  
            // Внимание - Для больших баз данных исполнение данного запроса может потребовать некоторого времени.  
            // О формировании запросов на языке XQuery - см. Статью 1 и Статью 2._            _if (ArrayCount (personArray) == 0) alert ('Объект не найден!');_  
  
            _A1 = ArrayOptFirstElem ( personArray ); // функция ищет первый элемент массива  
            alert (A1.fullname); // возвращает ФИО первого сотрудника из массива personArray  
            \_query\_str1 = "for $elem in collaborators where contains($elem/fullname, 'аааа') return $elem"; // получаем пустой массив  
            personArray1 = XQuery(\_query\_str1);  
            A2 = ArrayOptFirstElem (personArray1); // функция ищет первый элемент массива  
            alert (A2); // возвращает значение undefined, так как массив personArray1 не содержит ни одного элемента. В этом случае A2 не является элементом массива.  
      }  
      catch (err){   
            alert ('Объект не найден!');  
      };_

 

_Пример 2:_  
      _var Array1 = \[1, 2, 3\];  
      A1 = ArrayOptFirstElem ( Array1 );  
      alert ( A1 ); // возвращает значение 1_  
  
      _A2 = ArrayOptFirstElem ( \[\] ); // первый аргумент - пустой массив (массив без элементов), второй аргумент отсутствует  
      alert (A2); // возвращает undefined  
  
      A3 = ArrayOptFirstElem ( \[\], {id:0} ); // первый аргумент - пустой массив, второй аргумент - объект с одним свойством {id:0}  
      alert (A3.id); // возвращает 0 (значение свойства id)_  
  
      _A4 = ArrayOptFirstElem( \[\], 1 ); // выбирает из состава аргумента первый элемент массива (скалярный элемент 1)  
      alert (A4); // возвращает значение 1  
  
      A5 = ArrayOptFirstElem ( XQuery("for $obj in collaborators where $obj/fullname='Иванов Иван Иванович' return $obj"), {id:0} ); // первый аргумент - выборка из каталога, второй аргумент - объект с одним свойством {id:0}  
      alert (A5.id); // возвращает значение id сотрудника, полное имя которого - 'Иванов Иван Иванович', а если такой сотрудник отсутствует, то значение 0_

_Пример 3:_  
      _A1 = ArrayOptFirstElem ( undefined ); // выдается сообщение об ошибке с прерываем выполнения программного кода_
